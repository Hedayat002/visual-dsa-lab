class Solution:
  def dfs(self, grid, i, j, xor_value, m, n, dp):
    xor_value ^= grid[i][j]
    if i == m - 1 and j == n - 1:
      return xor_value
    if dp[i][j][xor_value] != -1:
      return dp[i][j][xor_value]
    ans = float('inf')
    if j + 1 < n:
      ans = min(ans, self.dfs(grid,i,j+1,xor_value,m,n,dp))
    if i + 1 < m:
      ans = min(ans, self.dfs(grid,i+1,j,xor_value,m,n,dp))
    dp[i][j][xor_value] = ans
    return dp[i][j][xor_value]
  def minCost(self, grid):
    m, n = len(grid), len(grid[0])
    dp = [[[-1]*1024 for _ in range(n)] for _ in range(m)]
    return self.dfs(grid,0,0,0,m,n,dp)
